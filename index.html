Integração

Compatível com o SDK
que sua equipe já usa

Troque apenas a URL base. Nada mais muda.

integracao.py 
from openai import OpenAI

# Apenas troque a base_url — o resto é idêntico
client = OpenAI(
    base_url="https://mangabarouter.online/api/v1",
    api_key="mangaba-sua-chave-aqui",
)

resposta = client.chat.completions.create(
    model="anthropic/claude-3.5-sonnet",
    messages=[
        {"role": "system", "content": "Você é um assistente jurídico."},
        {"role": "user",   "content": "Resuma os principais riscos deste contrato."},
    ],
    stream=True,
)

for chunk in resposta:
    print(chunk.choices[0].delta.content or "", end="")
