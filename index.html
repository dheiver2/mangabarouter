⚡ Integre em Minutos

API 100% compatível com OpenAI SDK — troque o endpoint e pronto

python · chat_completion.py 
from openai import OpenAI

client = OpenAI(
    base_url="https://openrouter.ai/api/v1",
    api_key="sk-or-sua-chave-aqui",
)

response = client.chat.completions.create(
    model="meta-llama/llama-3.1-8b-instruct:free",  # Grátis!
    messages=[
        {"role": "user", "content": "Olá! Qual é a capital do Brasil?"}
    ],
    extra_headers={
        "HTTP-Referer": "https://dheiver2.github.io/mangabarouter",
        "X-Title": "MangabaRouter"
    }
)

print(response.choices[0].message.content)
